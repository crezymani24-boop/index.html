    function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  if (payload.recordType === 'config') {
    var cfg = ss.getSheetByName('Config') || ss.insertSheet('Config');
    cfg.getRange('A1').setValue(JSON.stringify(payload.teamMembers || []));
    cfg.getRange('A2').setValue(JSON.stringify(payload.gpsRules || {}));
    return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
  }
  var sheetName = payload.recordType === 'attendance' ? 'Attendance' : 'Expenses';
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  if (payload.recordType === 'attendance') {
    sheet.appendRow([payload.id, payload.date, payload.staffId, payload.staffName, payload.inTime, payload.outTime || '', payload.minutes || 0, payload.hourlyRate || 0, payload.dayPay || 0, payload.distanceMeters ?? '']);
  } else {
    sheet.appendRow([payload.id, payload.date, payload.type, payload.tag, payload.member, payload.description, payload.amount]);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var expSheet = ss.getSheetByName('Expenses');
  var attSheet = ss.getSheetByName('Attendance');
  var cfgSheet = ss.getSheetByName('Config');
  var expenses = expSheet ? expSheet.getDataRange().getValues().slice(1).map(function(r) {
    return { id: Number(r[0]), date: String(r[1]), type: r[2], tag: r[3], member: r[4], description: r[5], amount: Number(r[6]) };
  }) : [];
  var attendance = attSheet ? attSheet.getDataRange().getValues().slice(1).map(function(r) {
    return { id: Number(r[0]), date: String(r[1]), staffId: r[2], staffName: r[3], inTime: String(r[4]), outTime: String(r[5]), minutes: Number(r[6]), hourlyRate: Number(r[7] || 0), dayPay: Number(r[8] || 0), distanceMeters: r[9] !== '' ? Number(r[9]) : null };
  }) : [];
  var teamMembers = cfgSheet && cfgSheet.getRange('A1').getValue() ? JSON.parse(cfgSheet.getRange('A1').getValue()) : null;
  var gpsRules = cfgSheet && cfgSheet.getRange('A2').getValue() ? JSON.parse(cfgSheet.getRange('A2').getValue()) : null;
  return ContentService.createTextOutput(JSON.stringify({ expenses: expenses, attendance: attendance, teamMembers: teamMembers, gpsRules: gpsRules })).setMimeType(ContentService.MimeType.JSON);
}
